# Jenkinsfile - Basic CO Pipeline
 Pipeline {
    agent any
    stage  {
         stage('Build') {
            steps {
                echo 'Building project'
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests'
            }
           
        }
        stage('Deploy') {
            steps {
               echo 'Deploying application'
            }
        }
     }
}

# GitLab CICD - Basic Pipleline
 stages:
     - build
     - test
     - deploy

build:
   stage: build
   script:
     - echo "Building projects"

 test:
   stage: test
   script:
     - echo "Running tests"

 deploy:
   stage: deploy
   script:
      - echo "Deploying apllication"

# Ansiable -Install and Configure Nginx
- name: Install and configure Nginx
  hosts: all
  become: yes
  tasks: 
    - name: Install ngnix
      yum:
       name: ngnix
       state: present
   - name: cpnfigure ngnix
     templete:
       src: ngnix.conf.j2
       dest: /etc/ngnix/ngnix.conf
   - name: start ngnix service
     service:
      name: ngnix
      state: started
      enable:yes
