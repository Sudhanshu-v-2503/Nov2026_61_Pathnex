# Ansible - Install Helm on Server
- name: Install Helm
  hosts:all
  become: yes
  tasks:
    - name: Download Helm
      get_url:
        url: https:// get.helm.sh/helm-v3.7.0-linux-amd64.tar.gz
        dest: /tmp/helm.tar.gz

    - name: Set Helm binary permissions
      file:
        path: /usr/local/bin/
        mode: '0755'

# Jenkinsfile -Multi-Stage Pipeline
pipeline {
      agent any
      environment {
         IMAGE_NAME = 'pathnex-web-app'
    }
    stages {
       stage('Checkout') {
        steps {
            git 'https://github.com/pathnex/sample-repo.git'
          }
      }
      stage('Build Docker Image') {
        steps {
          scripts {
              docker.withRagistry('https://docker.io', 'docker-credentials') {
                 docker.image("${env.IMAGE_NAME}").push('latest')
             }
         }
     }
  }
  Stage(''Deploy to Kubernetes') {
      steps {
        scriipt {
             sh 'kubectl apply -f deployment.yaml'
           }
       }
    }
  }
}

# GitLab CI/CD - Build, Push, and Deploy with Helm
stages:
   - build
   - push
   - deploy

build:
  stage: push
  script: 
     - docker build -t pathnex-web-app

push:
   stage: deploy
   script:
     - docker login -u "$SCI_REGISTORY_USER" -p "$CI_REGISTORY_PASSWORD"
     - docker push pathnex-web-app

deploy:
   stage: deploy
   script:
     - helm repo add stable https://charts.helm.sh/stable
     - helm install pathnex-ngnix stable/nginx-ingress
