Ansible - Install and Enable Ngnix
- name: Install and start Ngnix on Pathnex
  hosts: all
  become: yes

  tasks:
    - name:Install ngnix
      yum:
	name: ngnix
	state: present

    - name: Enable ngnix
      service:
	name: ngnix
	state: started
	enable: yes
